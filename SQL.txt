-- LEVEL 1 — Basic INNER JOIN

select * from student where course_id = 4;

select s.student_name, s.city, c.course_name from student s Inner join courses c on c.course_id = s.course_id;

select s.student_name, c.course_name as 'Course Name' from student s Inner join courses c On s.course_id = c.course_id  where c.course_name = "Java Full Stack";

select s.student_name, c.course_name from student s INNER JOIN courses c ON s.course_id = c.course_id where course_name = "Spring Boot";

select s.student_name, s.marks, c.course_name from student s inner join courses c On s.course_id = c.course_id ;

select s.student_name, s.city, c.course_name from student s Inner join courses c On s.course_id = c.course_id where s.city = "PUNE";

select s.student_name,s.marks, c.course_name from student s Inner JOIN courses c On s.course_id = c.course_id where s.marks > 80;

select s.student_name, s.marks , c.course_name from student s Inner JOIN courses c On s.course_id = c.course_id where s.student_name like'r%'

-- LEVEL 2 — WHERE + JOIN

select s.student_name,s.city, c.course_name from student s Inner JOin courses c On s.course_id = c.course_id where s.city='pune' And course_name = "Java Full Stack";

select s.student_name,s.marks,c.course_name from student s Inner join  courses c On s.course_id = c.course_id where s.marks > 75;

select s.student_name,s.marks,c.course_name from student s Inner join courses c On s.course_id = c.course_id where s.marks >= 70 AND s.marks <= 90; 

select s.student_name,c.fees, c.course_name from student s Inner join courses c On s.course_id = c.course_id where c.fees > 40000;

select s.student_name, c.trainer_name from student s Inner join courses c On s.course_id = c.course_id where c.trainer_name = "Rohit";

select s.student_name ,s.city, s.marks from student s Inner join courses c ON s.course_id = c.course_id where s.city = "Mumbai" And s.marks > 70;

-- LEVEL 3 — GROUP BY + JOIN + HAVING

select c.course_name, count(s.student_id) as 'Student count' from courses c Inner join student s On s.course_id = c.course_id group by course_name having count(s.student_id);

select c.course_name,count(s.student_id) AS 'Student Count'  from courses c Inner Join student s ON c.course_id = s.course_id Group by  course_name having count(s.student_id)>1;

select c.course_name,AVG(s.marks) as 'Average marks' from courses c Inner Join student s ON c.course_id = s.course_id group by c.course_name ;

select c.course_name, Avg(s.marks) AS 'Average Marks' from student s  Inner Join courses c On c.course_id = s.course_id Group by c.course_name Having Avg(s.marks)>80;

select c.course_name ,Max(s.student_id) as 'Max Student No' from student s Inner join courses c On c.course_id = s.course_id Group by c.course_name ;

select c.trainer_name,count(s.student_id) as 'Student count' from student s Inner join courses c ON c.course_id = s.course_id group by c.trainer_name

select c.trainer_name,count(s.student_id) as 'Student count' from student s Inner join courses c ON c.course_id = s.course_id group by c.trainer_name having count(s.student_id)>2;

select c.course_name,Max(s.marks) AS 'Highest Marks' from  student s Inner Join courses c On c.course_id = s.course_id group by c.course_name ;

-- LEVEL 4 — LEFT JOIN

select c.course_name,s.student_name  from student s Left Join courses c On c.course_id = s.course_id;

select c.course_name, count(s.student_id) as 'students No' from courses c left join student s On c.course_id = s.course_id Group By c.course_name having count(s.student_id)>=1;

select c.course_name, count(s.student_id) as 'student No' from courses c Left join student s On c.course_id = s.course_id group by c.course_name ;

-- LEVEL 5 — RIGHT JOIN

select c.course_name, s.student_name  from courses c right join student s On s.course_id = c.course_id ;

select s.student_name ,c.course_name from courses c Right join student s On s.course_id = c.course_id where c.course_name IS null ;
